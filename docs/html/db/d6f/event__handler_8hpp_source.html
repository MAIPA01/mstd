42 _MSTD_CONSTEXPR20 id_type add_callback(const event_action_handler& callback) {

61 _MSTD_CONSTEXPR20 void invoke(InvokeArgs&&... args) const {
62 if (_events.empty()) { return; }
63
64 // SAFETY WHEN CALLBACK DELETES ITSELF
65 std::vector<event_action_handler> callbacksToRun;
66 callbacksToRun.reserve(_events.size());
67 callbacksToRun.insert(callbacksToRun.end(), _events.begin(), _events.end());
68
69 for (const auto& callback : callbacksToRun) { callback(std::forward<InvokeArgs>(args)...); }
70 }
71
72 _MSTD_CONSTEXPR20 id_type operator+=(const event_action_handler& callback) { return add_callback(callback); }
73
74 _MSTD_CONSTEXPR20 bool operator-=(const id_type callbackId) { return remove_callback(callbackId); }

81 #endif
