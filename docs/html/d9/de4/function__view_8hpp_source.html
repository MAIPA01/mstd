71 template<class FreeFunc>
72 static Ret _free_func_invoker(void* func, void*, Args&&... args) noexcept(is_noexcept_function_v<FreeFunc>) {
73 using func_ptr = c_func_t<function_type_t<FreeFunc> >;
74
75 func_ptr freeFunc = reinterpret_cast<func_ptr>(func);
76 return std::invoke(freeFunc, std::forward<Args>(args)...);
77 }
78
79 template<class MemberFunc>
80 static Ret _member_func_invoker(void* func, void* parent,
81 Args&&... args) noexcept(is_noexcept_function_v<MemberFunc>) {
82 using parent_type = function_parent_t<MemberFunc>;
83 using parent_ptr = parent_type*;
84 using func_ptr = c_member_func_t<parent_type, function_type_t<MemberFunc> >;
85
86 parent_ptr parentObj = static_cast<parent_type*>(parent);
87 func_ptr memberFunc = *static_cast<const func_ptr*>(func);
88 return std::invoke(memberFunc, parentObj, std::forward<Args>(args)...);
89 }
90
91 template<class Functor>
92 static Ret _functor_invoker(void* func, void*, Args&&... args) noexcept(is_noexcept_function_v<Functor>) {
93 return std::invoke(*static_cast<Functor*>(func), std::forward<Args>(args)...);
94 }
95
96 using _invoker_type = Ret (*)(void*, void*, Args&&...) noexcept(is_noexcept_function_v<F>);

98 void* _func = nullptr;
